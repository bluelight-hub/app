### Screenshot Generation Strategy

[Source: docs/bmad/prd/lagekarte-prd.md#FR7: ETB-Integration]

**html2canvas Approach:**
```typescript
import html2canvas from 'html2canvas';

async function captureMapScreenshot(mapElement: HTMLElement): Promise<Blob> {
  // Capture map as canvas
  const canvas = await html2canvas(mapElement, {
    useCORS: true, // Enable cross-origin tiles
    scale: 2,      // 2x resolution for quality
  });

  // Convert canvas to blob
  return new Promise((resolve) => {
    canvas.toBlob((blob) => {
      resolve(blob as Blob);
    }, 'image/png');
  });
}
```

**Quality Considerations:**
- **Min Resolution:** 1024x768px (readable on desktop)
- **Scale Factor:** 2x (high-DPI displays)
- **Format:** PNG (lossless, supports transparency)
- **CORS:** Enable `useCORS` for OSM tiles

### File Upload Endpoint

[Source: docs/development/coding-standards.md#Backend (NestJS Modular)]

**Controller Implementation:**
```typescript
import { FileInterceptor } from '@nestjs/platform-express';
import { diskStorage } from 'multer';
import { extname } from 'path';

@Post('/einsatz/:einsatzId/lagekarte/screenshot')
@UseGuards(JwtAuthGuard)
@UseInterceptors(
  FileInterceptor('file', {
    storage: diskStorage({
      destination: './uploads/lagekarte',
      filename: (req, file, cb) => {
        const einsatzId = req.params.einsatzId;
        const timestamp = Date.now();
        // Sanitize: Only allow .png extension
        cb(null, `${einsatzId}_${timestamp}.png`);
      },
    }),
    limits: { fileSize: 2 * 1024 * 1024 }, // 2MB max
    fileFilter: (req, file, cb) => {
      // Validate MIME type: Only accept image/png
      if (file.mimetype !== 'image/png') {
        return cb(new Error('Only PNG files allowed'), false);
      }
      cb(null, true);
    },
  })
)
async uploadScreenshot(
  @Param('einsatzId') einsatzId: string,
  @UploadedFile() file: Express.Multer.File
) {
  return {
    url: `/uploads/lagekarte/${file.filename}`,
  };
}
```
