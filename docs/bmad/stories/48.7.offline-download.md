      - Example implementation:
        ```typescript
        import L from 'leaflet';
        import 'leaflet.offline';

        export function downloadTiles(
          baseLayer: L.TileLayer,
          bounds: L.LatLngBounds,
          zoomLevels: number[],
          onProgress: (progress: number) => void,
          onComplete: () => void,
          onError: (error: Error) => void
        ): L.Control.SaveTiles {
          const saveControl = L.control.savetiles(baseLayer, {
            zoomlevels: zoomLevels,
            confirm: null, // Skip confirmation dialog (we handle UI separately)
          });

          let totalTiles = 0;
          let savedTiles = 0;

          // Track download start
          saveControl.on('savestart', (event: any) => {
            totalTiles = event._tilesforSave?.length || 0;
            savedTiles = 0;
            onProgress(0);
          });

          // Track individual tile downloads
          saveControl.on('savetileend', () => {
            savedTiles++;
            const progress = totalTiles > 0 ? (savedTiles / totalTiles) * 100 : 0;
            onProgress(progress);
          });

          // Handle completion
          saveControl.on('saveend', () => {
            onProgress(100);
            onComplete();
          });

          // Handle errors
          saveControl.on('tileerror', (error: any) => {
            onError(new Error(error.error || 'Tile download failed'));
          });

          // Set bounds and trigger download programmatically
          saveControl.setBounds(bounds);
          saveControl._saveTiles(); // Trigger download (private method, but necessary for programmatic use)

          return saveControl;
        }
        ```
    - Track progress: Use event listeners (`savestart`, `savetileend`, `saveend`)

### Tile Download Logic

[Source: docs/bmad/features/lagekarte/epic-48-lagekarte.md#Dependencies & External Libraries]

**Tile Download Workflow (using leaflet.offline):**
1. User selects region (bounding box) and zoom levels (8-18) in OfflineRegionModal
2. Create save control with `L.control.savetiles(baseLayer, { zoomlevels: [8, 9, ..., 18] })`
3. Set bounds on control: `saveControl.setBounds(bounds)`
4. Trigger download programmatically: `saveControl._saveTiles()`
5. leaflet.offline handles:
   - Tile URL calculation automatically
   - Download via fetch()
   - Storage in IndexedDB with timestamp
6. Track progress via events:
   - `savestart`: Get total tile count
   - `savetileend`: Increment downloaded count
   - `saveend`: Download complete
   - `tileerror`: Handle failed downloads

**leaflet.offline handles tile calculation internally, no manual URL generation needed**

**Example Event-Based Download (see Task 7 for full implementation):**

### Storage Quota Check

[Source: docs/bmad/prd/lagekarte-prd.md#2.2 Non-Functional Requirements]

**Storage Manager API:**
```typescript
// utils/storage-quota.ts
export async function getStorageQuota(): Promise<{ used: number, available: number, percentage: number }> {
  if (!navigator.storage?.estimate) {
    throw new Error('Storage API not supported');
  }

  const estimate = await navigator.storage.estimate();
  const used = estimate.usage || 0;
  const total = estimate.quota || 0;
  const available = total - used;
  const percentage = (used / total) * 100;

  return {
    used: Math.round(used / (1024 * 1024)), // MB
    available: Math.round(available / (1024 * 1024)), // MB
    percentage: Math.round(percentage),
  };
}
```

### TTL Cleanup Strategy

[Source: docs/bmad/features/lagekarte/epic-48-lagekarte.md#Dependencies & External Libraries]

**30-Day TTL:**
- leaflet.offline stores tiles with `createdAt` timestamp automatically
- On app startup: Check tile timestamps via `getStorageInfo()`
- Delete tiles older than 30 days using `removeTile(key)`

**Cleanup Implementation (using leaflet.offline API):**
```typescript
// utils/offline-cleanup.ts
import { getStorageInfo, removeTile } from 'leaflet.offline';

export async function cleanupExpiredTiles(urlTemplate: string): Promise<void> {
  const allTiles = await getStorageInfo(urlTemplate);

  const now = Date.now();
  const ttl = 30 * 24 * 60 * 60 * 1000; // 30 days

  const expiredTiles = allTiles.filter(tile => {
    return (now - tile.createdAt) > ttl;
  });

  console.log(`Cleaning up ${expiredTiles.length} expired tiles`);

  for (const tile of expiredTiles) {
    await removeTile(tile.key);
  }

  console.log('Cleanup complete');
}

// Example usage:
// cleanupExpiredTiles('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png');
```
